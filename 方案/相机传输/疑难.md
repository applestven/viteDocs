### 配置/etc/vsftpd.conf

```bash
# =========================================================
# vsftpd - Sony 相机 FTP 上传配置
# =========================================================

# -------------------------
# FTP 监听
# -------------------------
listen=YES
listen_ipv6=NO

# -------------------------
# 禁止匿名登录
# -------------------------
anonymous_enable=NO

# -------------------------
# 允许本地 Linux 用户登录
# -------------------------
local_enable=YES

# 允许上传、创建目录、删除、重命名
write_enable=YES
# 设置ftp登录之后的根目录
local_root=/home/apple/ftp

# -------------------------
# 本地用户权限
# -------------------------

# 新建文件权限
file_open_mode=0666

# 新建目录权限：
# 0666 + umask 000 = 文件 666
# 目录会按照目录默认权限创建
local_umask=000

# -------------------------
# FTP 用户进入自己的 Home
# -------------------------

# apple 登录后默认进入：
# /home/apple
#
# 如果相机 FTP 目录填写 ftp
# 实际就是：
# /home/apple/ftp
chroot_local_user=NO

# -------------------------
# 被动模式
# -------------------------
pasv_enable=YES

# 被动端口范围
pasv_min_port=50000
pasv_max_port=50009

# 外网访问时告诉客户端使用这个公网 IP
pasv_address=139.199.192.179

# -------------------------
# 数据连接
# -------------------------

# 允许客户端使用被动模式
connect_from_port_20=YES

# -------------------------
# FTP 超时
# -------------------------

idle_session_timeout=600
data_connection_timeout=120

# -------------------------
# 本地用户上传
# -------------------------

# 防止某些 FTP 客户端上传后无法继续操作
allow_writeable_chroot=YES

# -------------------------
# 日志
# -------------------------

xferlog_enable=YES

# 记录详细 FTP 命令
# 用于排查 Sony 相机 MKD/CWD/STOR 问题
log_ftp_protocol=YES

# 日志文件
vsftpd_log_file=/var/log/vsftpd.log

# -------------------------
# 时间
# -------------------------
use_localtime=YES

# -------------------------
# UTF-8
# -------------------------
utf8_filesystem=YES
```

## 打印日志 

```bash
sudo tail -f /var/log/vsftpd.log
```
